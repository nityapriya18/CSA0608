arr = [45, 23, 67, 12, 89, 34, 56, 78, 90, 11,
       29, 73, 18, 64, 37]

keys = [73, 18, 100]

for key in keys:
    comparisons = 0
    found = False

    for i in range(len(arr)):
        comparisons += 1

        if arr[i] == key:
            print("\nKey:", key)
            print("Element found at position", i + 1)
            print("Number of comparisons =", comparisons)
            found = True
            break

    if found == False:
        print("\nKey:", key)
        print("Element not found")
        print("Number of comparisons =", comparisons)
